Experiment tag: HTML - experiment hero_copy

Tag type Custom HTML, trigger DOM Ready - All Pages. Copy the whole block, including the script tags.

<script>
(function () {
  var forced = (location.search.match(/[?&]ab=([AB])/) || [])[1];
  var saved = (document.cookie.match(/(?:^|; )ab_hero=([AB])/) || [])[1];
  var variant = forced || saved || (Math.random() < 0.5 ? 'A' : 'B');
  document.cookie = 'ab_hero=' + variant + '; path=/; max-age=2592000';
  if (variant === 'B') {
    var btn = document.querySelector('a[data-cta="hero"]');
    if (btn) { btn.textContent = 'Get your first bag for 1 kr'; }
  }
  window.dataLayer.push({
    event: 'experiment_assigned',
    experiment: { name: 'hero_copy', variant: variant }
  });
})();
</script>

What it does, line by line: read a forced variant from ?ab=A or ?ab=B in the address; otherwise read the cookie; otherwise flip a coin. Remember the choice in a cookie for 30 days. If B, change the hero button text. Tell the dataLayer which variant this visitor is in, so every later event on the page can carry it.
